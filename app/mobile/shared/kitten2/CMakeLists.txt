cmake_minimum_required(VERSION 3.22)
project(OpenReadKitten2 LANGUAGES C CXX)
if(NOT OPENREAD_KITTEN2_SOURCE)
  message(FATAL_ERROR "Set OPENREAD_KITTEN2_SOURCE to the pinned runtime checkout")
endif()
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${OPENREAD_KITTEN2_SOURCE}" llama)
add_library(OpenReadKitten2 STATIC runtime.cpp)
target_compile_features(OpenReadKitten2 PRIVATE cxx_std_17)
target_include_directories(OpenReadKitten2 PUBLIC include PRIVATE "${OPENREAD_KITTEN2_SOURCE}/tools/kitten-tts" "${OPENREAD_KITTEN2_SOURCE}/vendor/nlohmann")
target_link_libraries(OpenReadKitten2 PRIVATE llama sqlite3)
