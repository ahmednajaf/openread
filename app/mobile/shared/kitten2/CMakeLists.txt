cmake_minimum_required(VERSION 3.22)
project(OpenReadKitten2 LANGUAGES C CXX)
if(NOT OPENREAD_KITTEN2_SOURCE)
  message(FATAL_ERROR "Set OPENREAD_KITTEN2_SOURCE to the pinned runtime checkout")
endif()
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${OPENREAD_KITTEN2_SOURCE}" llama)
add_library(OpenReadKitten2 STATIC runtime.cpp)
target_compile_features(OpenReadKitten2 PRIVATE cxx_std_17)
target_include_directories(OpenReadKitten2 PUBLIC include PRIVATE "${OPENREAD_KITTEN2_SOURCE}/tools/kitten-tts" "${OPENREAD_KITTEN2_SOURCE}/vendor/nlohmann")
target_link_libraries(OpenReadKitten2 PRIVATE llama sqlite3)

option(OPENREAD_KITTEN2_TESTS "Build local model-backed recovery regression" OFF)
if(OPENREAD_KITTEN2_TESTS)
  # Redirect the getter only in this duplicate test object, never the product.
  add_library(Kitten2RecoveryRuntime OBJECT runtime.cpp)
  target_compile_features(Kitten2RecoveryRuntime PRIVATE cxx_std_17)
  target_compile_definitions(Kitten2RecoveryRuntime PRIVATE llama_get_logits_ith=openread_test_logits)
  target_include_directories(Kitten2RecoveryRuntime PRIVATE include "${OPENREAD_KITTEN2_SOURCE}/tools/kitten-tts" "${OPENREAD_KITTEN2_SOURCE}/vendor/nlohmann")
  target_link_libraries(Kitten2RecoveryRuntime PRIVATE llama sqlite3)
  add_executable(kitten2-recovery-check ../../../../dev/tests/kitten2-runtime-recovery.cpp $<TARGET_OBJECTS:Kitten2RecoveryRuntime>)
  target_compile_features(kitten2-recovery-check PRIVATE cxx_std_17)
  target_include_directories(kitten2-recovery-check PRIVATE include)
  target_link_libraries(kitten2-recovery-check PRIVATE llama sqlite3)
endif()
