cmake_minimum_required(VERSION 3.22.1)
project(openread_speech LANGUAGES C CXX)
set(ENGINE common mnemonics error ieee80 compiledata compiledict dictionary encoding intonation langopts numbers phoneme phonemelist readclause setlengths soundicon spect ssml synthdata synthesize tr_languages translate translateword voices wavegen speech espeak_api)
set(SOURCES openread_speech.cpp context_alignment.cpp)
foreach(name IN LISTS ENGINE)
  list(APPEND SOURCES "vendor/libespeak-ng/${name}.c")
endforeach()
foreach(name case categories ctype proplist scripts tostring)
  list(APPEND SOURCES "vendor/ucd-tools/src/${name}.c")
endforeach()
add_library(openread_speech STATIC ${SOURCES})
set_target_properties(openread_speech PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(openread_speech PUBLIC include PRIVATE vendor vendor/include vendor/include/compat vendor/ucd-tools/src/include)
target_compile_definitions(openread_speech PRIVATE LIBESPEAK_NG_EXPORT=1 PATH_ESPEAK_DATA="" ESPEAK_NG_API= N_PATH_HOME=1024)
target_compile_features(openread_speech PRIVATE cxx_std_17)
