cmake_minimum_required(VERSION 3.22.1)
project(openread_phonemizer LANGUAGES C CXX)
set_property(GLOBAL PROPERTY JOB_POOLS openread_compile=2 openread_link=1)
set(CMAKE_JOB_POOL_COMPILE openread_compile)
set(CMAKE_JOB_POOL_LINK openread_link)
set(SPEECH "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../shared/speech")
add_subdirectory("${SPEECH}" "${CMAKE_CURRENT_BINARY_DIR}/speech")
add_library(openread_phonemizer SHARED bridge.cpp)
target_link_libraries(openread_phonemizer PRIVATE openread_speech)
target_compile_features(openread_phonemizer PRIVATE cxx_std_17)
target_link_options(openread_phonemizer PRIVATE "-Wl,-z,max-page-size=16384")

add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../../../shared/qwen" "${CMAKE_CURRENT_BINARY_DIR}/qwen")
target_link_options(openread_qwen PRIVATE "-Wl,-z,max-page-size=16384")
